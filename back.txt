#include <iostream>
#include <fstream>
#include <string>
#include <vector>

using namespace std;

int main(){
    string relacion;
    cout << "Nombre de la relacion: ";
    cin >> relacion;
    string path = "C:\\megatron\\" + relacion + ".txt";

    bool newfile = false;
    ifstream verificar(path);
    if (verificar.good()){
        newfile = true;
        cout << "Agregando nueva tupla ->" << endl;
    }
    verificar.close();

    ofstream archivo(path, newfile ? ios::app : ios::out);
    if (archivo.is_open()) {
        cout << "Archivo " << (newfile ? "abierto" : "creado") << " exitosamente en: " << path << endl;
    } else {
        cout << "No se pudo crear/abrir el archivo." << endl;
        return 1;
    }

    vector<string> nombresParametros;
    int cantidad;
    
    if (!newfile) {
        cout << "Cuantos parametros tendra la relacion?: ";
        cin >> cantidad;
        
        cout << "Ingresa los nombres de los parametros:" << endl;
        for (int i = 0; i < cantidad; i++) {
            string newparametro;
            cout << "Parametro " << (i+1) << ": ";
            cin >> newparametro;
            nombresParametros.push_back(newparametro);
        }
    }

    vector<string> RegisParametros;
    cout << "\nIngresa los registros para cada parametro:" << endl;
    for (int i = 0; i < nombresParametros.size(); i++) {
        string valor;
        cout << nombresParametros[i] << ": ";
        cin >> valor;
        RegisParametros.push_back(valor);
    }
    for (int i = 0; i < RegisParametros.size(); i++) {
        archivo << RegisParametros[i];
        if (i < RegisParametros.size() - 1) {
            archivo << "#";
        }
    }
    archivo << endl;
    archivo.close();
    return 0;
}
